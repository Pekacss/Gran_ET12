    Usuario ||--o{ Plantilla : crea
    Equipo ||--o{ Jugador : tiene
    Posicion ||--o{ Jugador : clasifica
    Jugador ||--o{ Puntuacion : recibe
    Jugador ||--o{ PlantillaTitular : integra
    Jugador ||--o{ PlantillaSuplente : integra
    Plantilla ||--o{ PlantillaTitular : contiene
    Plantilla ||--o{ PlantillaSuplente : contiene

    Usuario {
        SMALLINT UNSIGNED Id PK
        VARCHAR(50) Nombre
        VARCHAR(50) Apellido
        VARCHAR(100) Email UK
        VARCHAR(64) Contrasena
    }

    Equipo {
        TINYINT UNSIGNED Id PK
        VARCHAR(50) Nombre UK
    }

    Posicion {
        TINYINT UNSIGNED Id PK
        VARCHAR(20) Nombre UK
    }

    Jugador {
        SMALLINT UNSIGNED Id PK
        TINYINT UNSIGNED IdEquipo FK
        TINYINT UNSIGNED IdPosicion FK
        VARCHAR(50) Nombre
        VARCHAR(50) Apellido
        VARCHAR(50) Apodo
        DATE FechaNacimiento
        DECIMAL(10,2) Cotizacion
    }

    Puntuacion {
        TINYINT UNSIGNED Fecha PK
        SMALLINT UNSIGNED IdJugador PK, FK
        DECIMAL(3,1) Puntaje
    }

    Plantilla {
        INT UNSIGNED Id PK
        SMALLINT UNSIGNED IdUsuario FK
        TINYINT UNSIGNED Fecha
        VARCHAR(60) Nombre
    }

    PlantillaTitular {
        INT UNSIGNED Id PK
        INT UNSIGNED IdPlantilla FK
        SMALLINT UNSIGNED IdJugador FK
    }
    PlantillaSuplente {
        INT UNSIGNED Id PK
        INT UNSIGNED IdPlantilla FK
        SMALLINT UNSIGNED IdJugador FK
    }
