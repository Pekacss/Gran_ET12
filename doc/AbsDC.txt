    class Usuario {
        +ushort Id
        +string Nombre
        +string Apellido
        +string Email
        +DateTime FechaNacimiento
        +bool Administrador
        +string Contraseña
        +EsValido() bool
        +Validar() void
        +ValidarAdministrador() void
        +EsAdministrador() bool
    }

    class Equipo {
        +byte Id
        +string Nombre
        +EsValido() bool
        +Validar() void
    }

    class Posicion {
        +byte Id
        +string Nombre
        +Arquero byte
        +Defensor byte
        +Mediocampista byte
        +Delantero byte
        +EsValida() bool
        +Validar() void
    }

    class Jugador {
        +ushort Id
        +string Nombre
        +string Apellido
        +string Apodo
        +DateTime FechaNacimiento
        +byte IdEquipo
        +decimal Cotizacion
        +byte IdPosicion
        +CotizacionMaxima decimal
        +EsValido() bool
        +Validar() void
    }

    class Plantilla {
        +int Id
        +ushort IdUsuario
        +byte Fecha
        +string Nombre
        +decimal PresupuestoMaximo
        +int CantidadMaximaJugadores
        +CantidadMaximaJugadoresPredeterminada int
        +List~Jugador~ Titulares
        +List~Jugador~ Suplentes
        +AgregarTitular(Jugador) void
        +AgregarSuplente(Jugador) void
        +ObtenerPresupuesto() decimal
        +PresupuestoValido() bool
        +CantidadJugadoresValida() bool
        +FormacionTitularValida() bool
        +EsValida() bool
        +PuntajeFecha(byte, List~Puntuacion~) decimal
    }

    note for Plantilla "Nombre y PresupuestoMaximo son anulables en el modelo."

    %% Relaciones de datos
    Usuario "1" --> "0..*" Plantilla : posee
    Equipo "1" --> "0..*" Jugador : tiene
    Posicion "1" --> "0..*" Jugador : clasifica
    Plantilla "1" --> "0..*" PlantillaTitular : contiene
    Plantilla "1" --> "0..*" PlantillaSuplente : contiene
    Jugador "1" --> "0..*" PlantillaTitular : integra
    Jugador "1" --> "0..*" PlantillaSuplente : integra
    Jugador "1" --> "0..*" Puntuacion : recibe
